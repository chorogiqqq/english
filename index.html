구글 스프레드시트 자동 연동

URL을 입력하고 활성화하면 운동 기록 저장 시 구글 스프레드시트 표로 자동 전송됩니다!

1분 구글 시트 연동 설치 방법

  1. 구글 드라이브에서 새 구글 스프레드시트를 만듭니다.
  2. 상단 메뉴 확장 프로그램 ➔ Apps Script를 클릭합니다.
  3. 아래 스크립트 코드를 복사하여 붙여넣고 저장합니다.
  4. 우측 상단 배포 ➔ 새 배포 ➔ 유형: 웹 앱 선택 후, 액세스 권한: 모든 사용자(Anyone)로 설정하여 배포합니다.
  5. 생성된 웹 앱 URL을 위 입력란에 붙여넣으면 끝!
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["날짜", "사용자이름", "카테고리", "운동종목", "세트수", "세트상세", "추정소모칼로리(kcal)", "메모"]);
  }
  
  var setsDetail = data.sets.map(function(s, idx) {
    return (idx + 1) + "세트: " + (s.weight ? s.weight + "kg x " + s.reps + "회" : s.duration + "분 " + s.distance + "km");
  }).join(" | ");
  
  sheet.appendRow([
    data.date,
    data.userName || "기본사용자",
    data.category,
    data.name,
    data.sets.length,
    setsDetail,
    data.calories || 0,
    data.memo || ""
  ]);
  
  return ContentService.createTextOutput("OK");
}
